cmake_minimum_required(VERSION 3.24)
project(strata_sycl_host_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
enable_testing()
add_executable(positioned_file_test positioned_file_test.cpp)
target_include_directories(positioned_file_test PRIVATE ../../include)
target_link_libraries(positioned_file_test PRIVATE Threads::Threads)
add_test(NAME positioned_file_test COMMAND positioned_file_test)
set_tests_properties(positioned_file_test PROPERTIES TIMEOUT 60)
